    void toast.offsetWidth;

    const icons = { success: '?', error: '?', warn: '?' };
    icon.textContent = icons[type] || '?';
    text.textContent = message;
    toast.classList.add(type);

    requestAnimationFrame(() => toast.classList.add('visible'));

    if (toastTimeout) clearTimeout(toastTimeout);
    toastTimeout = setTimeout(() => toast.classList.remove('visible'), duration);
}

let wakeLockSentinel = null;
let wakeLockEnabled = false;
let canvasStream = null;
let canvasInterval = null;

function isWakeLockSupported() {
    return 'wakeLock' in navigator && typeof navigator.wakeLock.request === 'function';
}

async function requestWakeLockAPI() {
    if (!isWakeLockSupported()) return false;
    try {
        wakeLockSentinel = await navigator.wakeLock.request('screen');
        wakeLockSentinel.addEventListener('release', () => {
            wakeLockSentinel = null;
            updateWakeStatus();
        });
        return true;
    } catch (e) {
        console.warn('[WakeLock API] refusé :', e.message);
        return false;
    }
}

async function releaseWakeLockAPI() {
    if (wakeLockSentinel) {
        try { await wakeLockSentinel.release(); } catch (e) {}
        wakeLockSentinel = null;
    }
}

function startCanvasStream() {
    if (canvasStream) return true;
    try {
        const canvas = document.getElementById('noSleepCanvas');
        const video = document.getElementById('noSleepVideo');
        if (!canvas || !video) return false;

        const ctx = canvas.getContext('2d');
        if (!ctx) return false;

        ctx.fillStyle = '#000000';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        if (typeof canvas.captureStream !== 'function') {
            console.warn('[Canvas Stream] captureStream non supporté');
            return false;
        }

        canvasStream = canvas.captureStream(1);

        let toggle = false;
        canvasInterval = setInterval(() => {
            toggle = !toggle;
            ctx.fillStyle = toggle ? '#000000' : '#010101';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }, 1000);

        video.srcObject = canvasStream;
        video.muted = true;
        video.playsInline = true;

        const playPromise = video.play();
        if (playPromise && playPromise.catch) {
            playPromise.catch(e => {
                console.warn('[Video fallback] Lecture refusée :', e.message);
            });
        }

        console.log('[Canvas Stream] Actif');
        return true;
    } catch (e) {
        console.warn('[Canvas Stream] Échec :', e.message);
        return false;
    }
}

function stopCanvasStream() {
    if (canvasInterval) {
        clearInterval(canvasInterval);
        canvasInterval = null;
    }
    const video = document.getElementById('noSleepVideo');
    if (video) {
        try {
            video.pause();
            video.srcObject = null;
        } catch (e) {}
    }
    if (canvasStream) {
        try {
            canvasStream.getTracks().forEach(t => t.stop());
        } catch (e) {}
        canvasStream = null;
    }
}

async function activateWakeLock() {
    let success = false;
    if (isWakeLockSupported()) {
        success = await requestWakeLockAPI();
    }
    if (!success) {
        success = startCanvasStream();
    }
    if (!success && !canvasStream) {
        success = startCanvasStream();
    }
    return success;
}

async function deactivateWakeLock() {
    await releaseWakeLockAPI();
    stopCanvasStream();
}

function updateWakeStatus() {
    const el = document.getElementById('wake-status');
    const txt = document.getElementById('wake-status-text');
    const btn = document.getElementById('wake-toggle-btn');
    if (!el || !txt || !btn) return;

    el.classList.remove('active', 'warn');

    const hasAnyMechanism = isWakeLockSupported() ||
        (document.getElementById('noSleepCanvas')?.captureStream);

    if (!hasAnyMechanism) {
        el.classList.add('warn');
        txt.textContent = '?? Non supporté par ce navigateur';
        btn.style.display = 'none';
        return;
    }

    btn.style.display = 'inline-block';

    const isActive = wakeLockSentinel !== null || canvasStream !== null;

    if (wakeLockEnabled && isActive) {
        el.classList.add('active');
        txt.textContent = '? L\'écran reste allumé';
        btn.textContent = 'Désactiver';
    } else if (wakeLockEnabled && !isActive) {
        el.classList.add('warn');
        txt.textContent = '? En attente (touchez la page)';
        btn.textContent = 'Désactiver';
    } else {
        txt.textContent = '?? Veille automatique';
        btn.textContent = 'Garder l\'écran allumé';
    }
}

async function toggleWakeLock() {
    if (wakeLockEnabled) {
        wakeLockEnabled = false;
        await deactivateWakeLock();
        try { localStorage.setItem(WAKE_LOCK_PREF, '0'); } catch (e) {}
        showToast('Veille automatique réactivée', 'warn');
    } else {
        wakeLockEnabled = true;
        const ok = await activateWakeLock();
        try { localStorage.setItem(WAKE_LOCK_PREF, '1'); } catch (e) {}

        if (ok) {
            showToast('L\'écran va rester allumé', 'success');
        } else {
            showToast('Activez manuellement dans les réglages', 'warn');
        }
    }
    updateWakeStatus();
}

document.addEventListener('visibilitychange', async () => {
    if (document.visibilityState === 'visible' && wakeLockEnabled) {
        await activateWakeLock();
        updateWakeStatus();
    }
});

async function ensureWakeLockActive() {
    if (!wakeLockEnabled) return;
    if (wakeLockSentinel === null && canvasStream === null) {
        await activateWakeLock();
        updateWakeStatus();
    }
}

function loadTimingCache() {
    try { return JSON.parse(localStorage.getItem(TIMING_CACHE_KEY) || '{}'); }
    catch (e) { return {}; }
}
function saveTimingCache(cache) {
    try { localStorage.setItem(TIMING_CACHE_KEY, JSON.stringify(cache)); } catch (e) {}
}
function recordGenerationTime(numFrames, durationMs) {
    const cache = loadTimingCache();
    const key = String(numFrames);
    if (!cache[key]) cache[key] = [];
    cache[key].push(durationMs);
    if (cache[key].length > 10) cache[key] = cache[key].slice(-10);
    saveTimingCache(cache);
    updateTimingDisplay();
}
function estimateGenerationTime(numFrames) {
    const cache = loadTimingCache();
    const samples = cache[String(numFrames)];
    if (samples && samples.length > 0) {
        const sorted = [...samples].sort((a, b) => a - b);
        const mid = Math.floor(sorted.length / 2);
        return sorted.length % 2 === 0 ? (sorted[mid-1] + sorted[mid]) / 2 : sorted[mid];
    }
    return 30000 + numFrames * 500;
}
function updateTimingDisplay() {
    const cache = loadTimingCache();
    const all = Object.values(cache).flat();
    const speedEl = document.getElementById('stat-speed');
    const samplesEl = document.getElementById('stat-samples');
    if (!speedEl || !samplesEl) return;
    if (all.length === 0) {
        speedEl.textContent = 'en attente de données';
        samplesEl.textContent = '0 génération';
        return;
    }
    const avg = all.reduce((a, b) => a + b, 0) / all.length;
    speedEl.textContent = (avg / 1000).toFixed(1) + ' s en moyenne';
    samplesEl.textContent = all.length + ' génération' + (all.length > 1 ? 's' : '');
}

function saveApiKey() {
    const input = document.getElementById('api-key-input');

        try { localStorage.removeItem(API_KEY_STORAGE); } catch (e) {}

        showToast('Clé supprimée', 'warn');
        return;
    }

    if (!key.startsWith('sk-')) {
        if (!confirm('La clé ne commence pas par "sk-". La sauvegarder quand même ?')) {

    try {
        localStorage.setItem(API_KEY_STORAGE, key);

        showToast('Clé API enregistrée ?', 'success');
        input.blur();
    } catch (e) {

function formatEta(seconds) {
    if (seconds < 60) return Math.round(seconds) + 's';
    const m = Math.floor(seconds / 60);
    const s = Math.round(seconds % 60);
    return m + 'm' + (s > 0 ? ' ' + s + 's' : '');
}

function fileToDataUri(file) {
    return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = e => resolve(e.target.result);
        reader.onerror = () => reject(new Error('Erreur lecture'));
        reader.readAsDataURL(file);
    });
}

function generateThumbnail(dataUri) {
    return new Promise((resolve) => {
        const img = new Image();
        img.onload = () => {
            try {
                const canvas = document.createElement('canvas');
                const ratio = img.height / img.width;
                canvas.width = THUMBNAIL_MAX_WIDTH;
                canvas.height = Math.round(THUMBNAIL_MAX_WIDTH * ratio);
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
                resolve(canvas.toDataURL('image/jpeg', 0.7));
            } catch (e) { resolve(dataUri); }
        };
        img.onerror = () => resolve(dataUri);
        img.src = dataUri;
    });
}

async function processOneFile(file) {
    if (!file.type.startsWith('image/')) return null;
    if (file.size > MAX_IMAGE_SIZE) {
        log('Image trop lourde ignorée : ' + file.name);
        return null;
    }
    try {
        const dataUri = await fileToDataUri(file);
        const thumbnail = await generateThumbnail(dataUri);
        return { file: file, dataUri: dataUri, thumbnail: thumbnail, id: Date.now() + Math.random() };
    } catch (e) {
        log('Erreur traitement ' + file.name + ' : ' + e.message);
        return null;
    }
}

async function handleFiles(files) {
    const imageFiles = Array.from(files).filter(f => f.type.startsWith('image/'));
    if (!imageFiles.length) return;

    state.isLoadingImages = true;
    updateGenerateBtn();
    setStatus('Chargement de ' + imageFiles.length + ' image' + (imageFiles.length > 1 ? 's' : '') + '…');

    const uploadZone = document.getElementById('upload-zone');
    uploadZone.classList.add('loading');

    const BATCH = 5;
    let processed = 0;

    for (let i = 0; i < imageFiles.length; i += BATCH) {
        const batch = imageFiles.slice(i, i + BATCH);
        const results = await Promise.allSettled(batch.map(f => processOneFile(f)));
        for (const r of results) {
            if (r.status === 'fulfilled' && r.value) {
                state.images.push(r.value);
                processed++;
            }
        }
        renderImages();
        setStatus('Chargement des images… ' + processed + '/' + imageFiles.length);
    }

    state.isLoadingImages = false;
    uploadZone.classList.remove('loading');
    setStatus(null);
    updateGenerateBtn();
}

function removeImage(id) {
    state.images = state.images.filter(img => img.id !== id);
    renderImages();
}

function renderImages() {
    const grid = document.getElementById('images-grid');

        return '<div class="image-card">' +
            '<img src="' + (img.thumbnail || img.dataUri) + '" alt="" loading="lazy">' +
            '<button class="remove-btn" data-remove="' + img.id + '">?</button>' +
            '</div>';
    }).join('');

    grid.querySelectorAll('[data-remove]').forEach(btn => {
        btn.addEventListener('click', (e) => {
            e.stopPropagation();
            removeImage(parseFloat(btn.dataset.remove));
        });
    });
}

        });
    });
}

// CONSTRUCTION DU PROMPT
// ------------------------------------------------------------------
function buildPrompt() {
    const effect = state.selectedEffect ? CARTOON_STYLES.find(s => s.id === state.selectedEffect) : null;
    const hasUserPrompt = state.mainPrompt && state.mainPrompt.trim().length > 0;

    const parts = [];

    // Base : transformation cartoon
    parts.push(
        'Animate this image as the starting frame. ' +
        'Transform the entire image into a cartoon / animation style. ' +
        'Keep the subject recognizable: same face, same pose, same composition as the source image. ' +
        'This is a stylized animation, not a re-imagination.'
    );

    // Style cartoon choisi
    if (effect) {
        parts.push(effect.prompt);
    } else {
        parts.push(
            'generic cartoon animation style, clean outlines, flat colors, expressive features. ' +
            'Transform the image into a cartoon style.'
        );
    }

    // Direction artistique utilisateur
    if (hasUserPrompt) {
        parts.push('Additional artistic direction (applied on top of the cartoon style): ' + state.mainPrompt.trim());
    }

    // Cadrage
    if (state.framing !== 'random' && FRAMING_PRESETS[state.framing]) {
        parts.push(FRAMING_PRESETS[state.framing]);
    }

    // Intensité
    if (state.intensity !== 'random' && INTENSITY_PRESETS[state.intensity]) {
        parts.push(INTENSITY_PRESETS[state.intensity]);
    } else {
        parts.push('moderate cartoon movement, clear and expressive');
    }

    // Action, caméra, lumière
    parts.push(randomItem(RANDOM_ACTIONS));
    parts.push(randomItem(RANDOM_CAMERA));
    parts.push(randomItem(RANDOM_LIGHTING));

    parts.push('Sound design: playful cartoon foley, soft ambient textures, no music, no dialogue');
    parts.push('Portrait orientation 9:16 vertical, 24fps, no text, no subtitles, no watermarks, consistent cartoon character design with the source image');

    return parts.join('. ');
}

// API AGNES
// ------------------------------------------------------------------
async function apiFetch(url, options = {}, label = 'API') {
    for (let attempt = 0; attempt < 7; attempt++) {
        if (state.stopRequested) throw new Error('Arrêt demandé');
        try {
            const res = await fetch(url, options);
            if (res.status === 429) {
                state.creationAttemptsSinceLastError = 0;
                if (createIntervalMs < CREATE_INTERVAL_SAFE) {
                    createIntervalMs = Math.min(createIntervalMs + 8000, CREATE_INTERVAL_MAX);
                    log('429 détecté ? intervalle porté à ' + (createIntervalMs / 1000) + 's');
                }
                const wait = [15, 30, 45, 60, 90, 120, 180][Math.min(attempt, 6)];
                log(label + ' : 429, attente ' + wait + 's');
                setStatus('L\'intelligence artificielle se prépare… (' + wait + 's)');
                await sleep(wait * 1000);
                continue;
            }
            if (res.status === 503) {
                const wait = [5, 10, 15, 20, 30, 45][Math.min(attempt, 5)];
                log(label + ' : 503, attente ' + wait + 's');
                setStatus('Votre vidéo cartoon se prépare… (' + wait + 's)');
                await sleep(wait * 1000);
                continue;
            }
            if (res.ok && createIntervalMs > CREATE_INTERVAL_MIN) {
                state.creationAttemptsSinceLastError++;
                if (state.creationAttemptsSinceLastError >= 3) {
                    createIntervalMs = Math.max(createIntervalMs - 4000, CREATE_INTERVAL_MIN);
                    state.creationAttemptsSinceLastError = 0;
                    log('3 succès consécutifs ? intervalle ramené à ' + (createIntervalMs / 1000) + 's');
                }
            }
            return res;
        } catch (e) {
            if (e.message === 'Arrêt demandé') throw e;
            const wait = [3, 5, 8, 12, 20, 30][Math.min(attempt, 5)];
            log(label + ' : réseau, retry ' + wait + 's');
            setStatus('Connexion en cours… (' + wait + 's)');
            await sleep(wait * 1000);
        }
    }
    return fetch(url, options);
}

async function createVideoTask(imageDataUri, prompt) {
    const body = {
        model: MODEL_VIDEO,
        prompt: prompt,
        image: imageDataUri,
        num_frames: state.durationFrames,
        frame_rate: FRAME_RATE
    };

    const res = await apiFetch(API_BASE + '/videos', {
        method: 'POST',
        headers: {
            'Authorization': 'Bearer ' + getApiKey(),
            'Content-Type': 'application/json'
        },
        body: JSON.stringify(body)
    }, 'Création');

    if (!res.ok) {
        const err = await res.text();
        throw new Error('HTTP ' + res.status + ' — ' + err.slice(0, 200));
    }

    const data = await res.json();
    const videoId = data.video_id || data.id || data.task_id;
    if (!videoId) throw new Error('Pas de video_id retourné');
    return videoId;
}

async function pollVideo(videoId, onProgress) {
    const estimatedMs = estimateGenerationTime(state.durationFrames);
    const estimatedSec = estimatedMs / 1000;

    let remaining = Math.max(20, Math.floor(estimatedSec * 0.8));
    while (remaining > 0) {
        if (state.stopRequested) throw new Error('Arrêt demandé');
        onProgress('Votre vidéo cartoon se prépare… — ' + remaining + 's');
        await sleep(1000);
        remaining--;
    }

    for (let attempt = 0; attempt < MAX_POLL_ATTEMPTS; attempt++) {
        if (state.stopRequested) throw new Error('Arrêt demandé');
        if (attempt > 0) {
            let r = POLL_INTERVAL_SEC;
            while (r > 0) {
                if (state.stopRequested) throw new Error('Arrêt demandé');
                onProgress('L\'image devient dessin animé… — ' + r + 's');
                await sleep(1000);
                r--;
            }
        }

        const url = POLL_BASE + '?video_id=' + encodeURIComponent(videoId) + '&model_name=' + encodeURIComponent(MODEL_VIDEO);
        const res = await apiFetch(url, {
            method: 'GET',
            headers: { 'Authorization': 'Bearer ' + getApiKey() }
        }, 'Polling');

        const d = await res.json();
        const status = d.status || 'unknown';
        const progress = d.progress || 0;
        onProgress('Transformation cartoon… — ' + progress + '%');

        if (status === 'completed' || status === 'succeeded' || status === 'done') {
            const videoUrl = (d.metadata && d.metadata.url) || d.url || (d.output && d.output.url);
            if (!videoUrl) throw new Error('Terminé mais pas d\'URL');
            return videoUrl;
        }
        if (status === 'failed' || status === 'error' || status === 'cancelled') {
            throw new Error('Échec (' + status + ')');
        }
    }
    throw new Error('Délai maximal dépassé');
}

// ------------------------------------------------------------------
// FILE D'ATTENTE
// ------------------------------------------------------------------
function renderQueue() {
    const queueEl = document.getElementById('queue');
    if (!state.queue.length) { queueEl.style.display = 'none'; return; }
    queueEl.style.display = 'block';

    const total = state.queue.length;
    const done = state.completed + state.failed;
    const inProgress = state.queue.filter(q => q.status === 'processing').length;
    const pending = total - done - inProgress;

    const estGen = estimateGenerationTime(state.durationFrames) / 1000;
    const remainingCreates = pending;
    const eta = remainingCreates > 0
        ? (remainingCreates * (createIntervalMs / 1000)) + estGen
        : (inProgress * estGen / Math.max(inProgress, 1));

    let html = '<div class="queue-summary">' +
        '<span class="progress-text">Progression : ' + done + '/' + total + '</span>' +
        '<span class="eta-text">' + (done < total ? '˜ ' + formatEta(eta) + ' restant' : 'Terminé') + '</span>' +
        '</div>';

    const visible = state.queue.slice(0, MAX_VISIBLE_QUEUE);
    const hidden = state.queue.length - visible.length;

    html += visible.map(item => {
        let icon = '?', statusText = 'En file d\'attente', cls = 'pending';
        if (item.status === 'creating') { icon = '?'; statusText = 'Préparation…'; cls = 'creating'; }
        else if (item.status === 'processing') { icon = '?'; statusText = item.progress || 'En cours…'; cls = 'processing'; }
        else if (item.status === 'done') { icon = '?'; statusText = 'Terminé'; cls = 'done'; }
        else if (item.status === 'failed') { icon = '?'; statusText = 'Échec'; cls = 'failed'; }

        return '<div class="queue-item ' + cls + '">' +
            '<div class="q-icon">' + icon + '</div>' +
            '<div class="q-name">Image ' + (item.imageIndex + 1) + '</div>' +
            '<div class="q-status">' + esc(statusText) + '</div>' +
        '</div>';
    }).join('');

    if (hidden > 0) {
        html += '<div class="queue-more">… et ' + hidden + ' autre' + (hidden > 1 ? 's' : '') + ' image' + (hidden > 1 ? 's' : '') + '</div>';
    }

    queueEl.innerHTML = html;
}

// ------------------------------------------------------------------
// PIPELINE
// ------------------------------------------------------------------
async function createTaskOnly(item) {
    item.status = 'creating';
    item.progress = 'Préparation…';
    renderQueue();

    const prompt = buildPrompt();
    item.prompt = prompt;

    log('[Image ' + (item.imageIndex + 1) + '] Création…');
    const videoId = await createVideoTask(item.image.dataUri, prompt);
    item.videoId = videoId;
    item.startTime = Date.now();
    item.status = 'processing';
    item.progress = 'En cours…';
    renderQueue();

    log('[Image ' + (item.imageIndex + 1) + '] Tâche créée : ' + String(videoId).slice(0, 30) + '…');
    return videoId;
}

async function pollTaskAndDisplay(item) {
    try {
        const videoUrl = await pollVideo(item.videoId, (prog) => {
            item.progress = prog;
            renderQueue();
        });

        const elapsed = Date.now() - item.startTime;
        recordGenerationTime(state.durationFrames, elapsed);
        log('[Image ' + (item.imageIndex + 1) + '] Terminée en ' + (elapsed / 1000).toFixed(1) + 's');

        item.status = 'done';
        item.videoUrl = videoUrl;
        item.progress = 'Terminé';
        state.completed++;
        renderQueue();
        updateGenerateBtn();

        addToGallery(videoUrl, 'Image ' + (item.imageIndex + 1));
    } catch (e) {
        if (e.message === 'Arrêt demandé') {
            item.status = 'failed';
            item.progress = 'Arrêté';
        } else {
            log('[Image ' + (item.imageIndex + 1) + '] ' + e.message);
            item.status = 'failed';
            item.progress = 'Échec';
            state.failed++;
        }
        renderQueue();
        updateGenerateBtn();
    }
}

// WORKFLOW PRINCIPAL
// ------------------------------------------------------------------
async function startGeneration() {
    if (state.isRunning || !state.images.length) return;
    if (!getApiKey()) {
        showToast('Ajoutez d\'abord votre clé API', 'error');
        document.getElementById('api-key-input').focus();
        return;
    }

    state.mainPrompt = document.getElementById('main-prompt').value.trim();

    if (!state.selectedEffect) {
        if (!confirm('Aucun style cartoon sélectionné. Continuer avec un style cartoon générique ?')) {
            return;
        }
    }

    state.isRunning = true;
    state.stopRequested = false;
    state.completed = 0;
    state.failed = 0;
    state.creationAttemptsSinceLastError = 0;

    createIntervalMs = CREATE_INTERVAL_MIN;

    state.queue = state.images.map((img, i) => ({
        imageIndex: i,
        image: img,
        status: 'pending',
        progress: null,
        videoId: null,
        videoUrl: null,
        prompt: null,
        startTime: null
    }));

    document.getElementById('gallery-grid').innerHTML = '';
    document.getElementById('gallery').style.display = 'none';

    document.getElementById('stop-btn').classList.add('visible');
    document.getElementById('generate-btn').disabled = true;

    await ensureWakeLockActive();

    renderQueue();
    updateGenerateBtn();

    log('Démarrage Cartoon Factory — ' + state.images.length + ' vidéos');
    log('Style : ' + (state.selectedEffect || 'générique'));
    log('Prompt : ' + (state.mainPrompt ? '"' + state.mainPrompt.slice(0, 60) + '…"' : '(vide)'));

    const totalStart = Date.now();
    const pollingTasks = [];

    try {
        for (let i = 0; i < state.queue.length; i++) {
            if (state.stopRequested) break;

            const item = state.queue[i];

            try {
                await createTaskOnly(item);
                pollingTasks.push(pollTaskAndDisplay(item));
            } catch (e) {
                if (e.message === 'Arrêt demandé') break;
                log('[Image ' + (i + 1) + '] Échec création : ' + e.message);
                item.status = 'failed';
                item.progress = 'Échec';
                state.failed++;
                renderQueue();
                updateGenerateBtn();
            }

            if (i < state.queue.length - 1 && !state.stopRequested) {
                const waitSec = Math.ceil(createIntervalMs / 1000);
                for (let r = waitSec; r > 0; r--) {
                    if (state.stopRequested) break;
                    const nextNum = i + 2;
                    setStatus('Respiration créative… — Image ' + nextNum + '/' + state.queue.length + ' dans ' + r + 's');
                    await sleep(1000);
                }
            }
        }

        if (pollingTasks.length > 0) {
            setStatus('Finalisation…');
            await Promise.allSettled(pollingTasks);
        }
    } catch (e) {
        log('Erreur globale : ' + e.message);
    }

    const totalElapsed = (Date.now() - totalStart) / 1000;
    const summary = state.stopRequested
        ? 'Arrêté — ' + state.completed + '/' + state.queue.length + ' vidéos'
        : '? ' + state.completed + '/' + state.queue.length + ' vidéos cartoon créées en ' + formatEta(totalElapsed) + ' ?';

    log(summary);
    setStatus(summary);
    await sleep(4000);
    setStatus(null);

    state.isRunning = false;
    state.stopRequested = false;
    document.getElementById('stop-btn').classList.remove('visible');
    updateGenerateBtn();

    showToast(state.completed + ' vidéo cartoon' + (state.completed > 1 ? 's' : '') + ' créée' + (state.completed > 1 ? 's' : ''), 'success');
}

function stopGeneration() {
    if (!state.isRunning) return;
    state.stopRequested = true;
    log('Arrêt demandé par l\'utilisateur');
    setStatus('Arrêt en cours…');
}
